cmake_minimum_required(VERSION 3.22)
project(RE_SYNTH_Portable_DSP_Tests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
foreach(name test_color_math test_spectral_dsp test_pitch_detector test_peak_tracking test_reconstruction_qa)
    add_executable(${name} ${name}.cpp)
    if(MSVC)
        target_compile_options(${name} PRIVATE /W4)
    else()
        target_compile_options(${name} PRIVATE -Wall -Wextra -Wpedantic)
    endif()
    add_test(NAME ${name} COMMAND ${name})
endforeach()
add_executable(benchmark_spectral benchmark_spectral.cpp)
